> **Usage Example**
> ```zn
> use std.collections.map;
> 
> fn main() {
>   let map = HashMap#[String, i32].new();
> 
>   map.insert(String.from("a"), 1);
> 
>   switch (
>     map.get(&String.from("a"))
>   ) {
>     .Some(value) => @println("{}", *value),
>     .None => @println("Nothing"),
>   };
> }
> ```
