> **Usage Example**
> ```zn
> use std.path;
>
> fn main() {
>   let p: []const char = "/home/user/main.zn";
>
>   @println("base: {}", Path.basename(&p));
>   @println("dir: {}", Path.dirname(&p));
>   @println("{}", Path.join("/usr/local", "bin"));
>   @println("{}", Path.normalize("/usr/./local/../bin"));
> }
> ```
